<p>代理模式+策略模式，典型 Spring 中的 DispatcherServlet</p>
<p>用户将任务全权委派给代理负责，代理负责根据策略任务调度具体的执行单元</p>
<h2 id="执行单元"><a class="headerlink" href="#执行单元" title="执行单元"></a>执行单元</h2><pre><code>/**
 * 执行单元实现的接口
 */
public interface ITarget {

<pre><code>void doSomething(String command);
</code></pre>
<p>}</p>
<p>/**</p>
<ul>
<li><p>执行单元A<br/> */<br/>public class TargetA implements ITarget {<br/> @Override<br/> public void doSomething(String command) {</p>
<p> }</p>
</li>
</ul>
<p>}</p>
<p>/**</p>
<ul>
<li><p>执行单元B<br/> */<br/>public class TargetB implements ITarget {<br/> @Override<br/> public void doSomething(String command) {</p>
<p> }</p>
</li>
</ul>
<p>}<br/></p></code></pre>
<h2 id="执行单元调度者"><a class="headerlink" href="#执行单元调度者" title="执行单元调度者"></a>执行单元调度者</h2><pre><code>public class Manager {
    private Map&lt;String ,ITarget&gt; targets = new HashMap&lt;String ,ITarget&gt;();

<pre><code>public Manager(){
    targets.put("commandA",new TargetA());
    targets.put("commandB",new TargetB());
}

public void dispatch(String command){
    targets.get(command).doSomething(command);
}
</code></pre>
<p>}<br/></p></code></pre>
<h2 id="使用者"><a class="headerlink" href="#使用者" title="使用者"></a>使用者</h2><pre><code>public class Boss {
    
<pre><code>public static void main(String[] args) {
    Manager manager = new Manager();
    manager.dispatch("Command1");
}
</code></pre>
<p>}<br/></p></code></pre>
